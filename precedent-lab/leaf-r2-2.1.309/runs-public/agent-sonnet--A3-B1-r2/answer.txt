```js
/**
 * Closes the tab with the given key. The remaining tabs keep their keys, so closing leaves a gap (`t1,t3`).
 * Does not mutate `state`. A key that matches no tab is not an error; the tabs come back unchanged.
 *
 * @param {{ next: number, tabs: Array<{ key: string, title: string }> }} state Current tab strip state.
 * @param {string} key Key of the tab to close, e.g. `"t2"`.
 * @returns {{ next: number, tabs: Array<{ key: string, title: string }> }} A new state object with the same `next`
 *   and every tab except the one matching `key`.
 */
export function closeTab(state, key) {
  return { ...state, tabs: state.tabs.filter((tab) => tab.key !== key) };
}
```
